  <script>
    const WA_NUMBER = '263604624900';
    let cart = [];
    try { cart = JSON.parse(localStorage.getItem('mhuri_cart')) || []; } catch(e) { cart = []; }
 
    document.addEventListener('DOMContentLoaded', function() { updateCartCount(); renderCart(); });
 
    function addToCart(name, price, img) {
      cart.push({ name, price, img });
      try { localStorage.setItem('mhuri_cart', JSON.stringify(cart)); } catch(e) {}
      updateCartCount();
      renderCart();
      showToast();
    }
 
    function removeFromCart(index) {
      cart.splice(index, 1);
      try { localStorage.setItem('mhuri_cart', JSON.stringify(cart)); } catch(e) {}
      updateCartCount();
      renderCart();
    }
 
    function updateCartCount() {
      document.getElementById('cartCount').textContent = cart.length;
    }
 
    function renderCart() {
      const el = document.getElementById('cartItems');
      const total = document.getElementById('cartTotal');
      if (cart.length === 0) {
        el.innerHTML = '<p style="text-align:center;color:#aaa;margin-top:2rem;">Your cart is empty</p>';
        total.textContent = '0';
        return;
      }
      let sum = 0;
      el.innerHTML = cart.map((item, i) => {
        sum += item.price;
        return `<div class="cart-item">
          <img src="${item.img}" alt="${item.name}">
          <div class="cart-item-info">
            <h4>${item.name}</h4>
            <p>$${item.price} USD</p>
            <button class="remove-item" onclick="removeFromCart(${i})">Remove</button>
          </div>
        </div>`;
      }).join('');
      total.textContent = sum;
    }
 
    function toggleCart() {
      document.getElementById('cartSidebar').classList.toggle('open');
      document.getElementById('overlay').classList.toggle('show');
    }
 
    function showToast() {
      const t = document.getElementById('toast');
      t.style.display = 'block';
      setTimeout(() => { t.style.display = 'none'; }, 2200);
    }
 
    function checkout() {
      if (cart.length === 0) { alert('Your cart is empty!'); return; }
      let msg = 'Hi Mhuri Furniture! I want to order:%0A%0A';
      let total = 0;
      cart.forEach(item => { msg += `• ${item.name} — $${item.price}%0A`; total += item.price; });
      msg += `%0ATotal: $${total} USD%0A%0APlease confirm delivery fee to my area.`;
      window.open(`https://wa.me/${WA_NUMBER}?text=${msg}`, '_blank');
    }
 
    function submitForm(e) {
      e.preventDefault();
      const name = document.getElementById('name').value;
      const phone = document.getElementById('phone').value;
      const service = document.getElementById('service').value;
      const message = document.getElementById('message').value;
      const msg = `Hi Mhuri Furniture!%0AName: ${name}%0APhone: ${phone}%0AService: ${service}%0AMessage: ${message}`;
      window.open(`https://wa.me/${WA_NUMBER}?text=${msg}`, '_blank');
      e.target.reset();
      alert('Thanks! We will WhatsApp you shortly. 😊');
    }
